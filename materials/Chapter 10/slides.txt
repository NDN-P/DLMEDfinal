Class Net
                      เป็นไฟล์ที่ใช้กําหนดโครงสร้างของแบบจําลอง
class Net(Module):   
    def __init__(self): # กําหนดโครงสร้างในConstructor
        super(Net, self).__init__()
        self.cnn_layers = Sequential(
            # Defining a 2D convolution layer
             Conv2d(1,4,kernel_size=3, stride=1, padding=1),
            BatchNorm2d(4),
            ReLU(inplace=True),
            MaxPool2d(kernel_size=2, stride=2),
            # Defining another 2D convolution layer
            Conv2d(4,4,kernel_size=3, stride=1, padding=1),
            BatchNorm2d(4),
           ReLU(inplace=True),
           MaxPool2d(kernel_size=2, stride=2),
        )
        self.linear_layers = Sequential(
            Linear(4 * 7 * 7, 10)
        )
    # Defining the forward pass    
    def forward(self, x):
        x = self.cnn_layers(x)
        x = x.view(x.size(0), -1)
        x = self.linear_layers(x)
        return x

Class Net

class ConvNet(Module):
    def __init__(self):
        super(ConvNet, self).__init__()
        self.layer1 = Sequential(
            Conv2d(1, 32, kernel_size=5, stride=1, padding=2),
            ReLU(),
            MaxPool2d(kernel_size=2, stride=2))
        self.layer2 = Sequential(
            Conv2d(32, 64, kernel_size=5, stride=1, padding=2),
            ReLU(),
            MaxPool2d(kernel_size=2, stride=2))
        self.drop_out = Dropout()
        self.fc1 = Linear(7 * 7 * 64, 1000)
        self.fc2 = Linear(1000, 10)
    def forward(self, x):
        out = self.layer1(x)
        out = self.layer2(out)
out = out.reshape(out.size(0), -1)
        out = self.drop_out(out)
        out = self.fc1(out)
        out = self.fc2(out)
        return out
model = ConvNet()
torch.save(model, 'C:/ConvNet.pt')

2D Convolution Layer
หมายถึง ส่วนที่มีหน้าที่สกัดคุณลักษณะที่สําคัญจากภาพ โดยอาศัยตัวดําเนินการคอนโวลูชั่นรวมกับตัวกรองที่ค่าสัมประสิทธิทุกค่าบนตัวกรองจะมีการปรับ
เหมาะอัตโนมัติตามค่าความผิดพลาดที่เกิดขึ้นจากแบบจําลอง
อ้างอิง https://syncedreview.com/2017/05/11/a-guide-to-receptive-field-arithmetic-for-convolutional-neural-networks/
คําอธิบายตัวแปร
• in_channels (int) คือ จํานวน channels ของภาพนําเข้า (input image)
(ภาพสีมี 3 channels ขณะที่ภาพระดับสีเทาและภาพขาวดํามี 1 channel)
• out_channels (int) คือ จํานวน channels ของผลลัพธ์ที่ผ่านการ conv
• kernel_size (int or tuple) คือ ขนาดของตัวกรอง conv
• stride (int or tuple) คือ จํานวนช่องในการขยับตัวกรอง 
• padding (int or tuple) คือ ระบุจํานวนชั้นที่เติมข้อมูล 
• padding_mode (string) คือ วิธีการเติมข้อมูล
torch.nn.Conv2d(
in_channels, out_channels, kernel_size, 
stride=1, padding=0, dilation=1, groups=1, 
bias=True, padding_mode='zeros’
)
หมายเหตุ สีเหลือง คือ บังคบต้องระบุ

2D Batch Normalization
หมายถึง ส่วนที่ทําหน้าที่ (1) ปรับช่วงค่าข้อมูลให้อยู่ในมาตรฐานเดียวกัน หรือ (2) อยู่ในขอบเขตที่กําหนด หรือ (3) ปรับค่าอยู่ในช่วงเดียวกัน (Scaling Data) 
ก่อนออกจาก Layer ไปเป็น input ให้ Layer ถัดไป ซึ่งสามารถ (1) ลดอิทธิพลของ feature ไม่ให้ตัวใดตัวหนึ่งมีอิทธิพลมากเกินไป (2) สามารถช่วยเพิ่ม
ความเร็วในการฝึกสอนแบบจําลอง และ (3) ทําให้ค่า Loss ลดลงได้มากกว่า

• num_features คือ จํานวน channels ของ input ใน Layer นี้

track_running_stats=True
 )

class Net(Module):   
    def __init__(self): # กําหนดโครงสร้างในConstructor
        super(Net, self).__init__()
        self.cnn_layers = Sequential(
            # Defining a 2D convolution layer
             Conv2d(1,4,kernel_size=3, stride=1, padding=1),
            BatchNorm2d(4),
            ReLU(inplace=True),
            MaxPool2d(kernel_size=2, stride=2),
            # Defining another 2D convolution layer
            Conv2d(4,4,kernel_size=3, stride=1, padding=1),
            BatchNorm2d(4),
           ReLU(inplace=True),
           MaxPool2d(kernel_size=2, stride=2),
        )
        self.linear_layers = Sequential(
            Linear(4 * 7 * 7, 10)
        )
    # Defining how the data flows through these layers when 
       performing the forward pass through the network
    def forward(self, x):
        x = self.cnn_layers(x)
      # convert feature map to vector form
        x = x.view(x.size(0), -1) 
        x = self.linear_layers(x)
        return x

class AlexNet(nn.Module):
    def __init__(self, num_classes: int = 1000, dropout: float = 0.5) -> None:
        super().__init__()
        _log_api_usage_once(self)
        self.features = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=11, stride=4, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
            nn.Conv2d(64, 192, kernel_size=5, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
            nn.Conv2d(192, 384, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(384, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
        )
        self.avgpool = nn.AdaptiveAvgPool2d((6, 6))
        self.classifier = nn.Sequential(
            nn.Dropout(p=dropout),
            nn.Linear(256 * 6 * 6, 4096),
            nn.ReLU(inplace=True),
            nn.Dropout(p=dropout),
            nn.Linear(4096, 4096),
            nn.ReLU(inplace=True),
            nn.Linear(4096, num_classes),
        )
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.features(x)
        x = self.avgpool(x)
        x = torch.flatten(x, 1)
        x = self.classifier(x)
        return x

ฟังก์ชัน model.train() ใน PyTorch 
model.train() ใน PyTorch เป็นคําสั่งที่ใช้เพื่อกําหนดว่าโมเดลกําลังอยู่ในโหมด "training" (การฝึกฝน) แทนที่จะเป็นโหมด "evaluation" (การ
ทดสอบ) การสลับโหมดระหว่างการฝึกฝนและการทดสอบเป็นสิ่งสําคัญ เนื่องจากพฤติกรรมของบางเลเยอร์ในโมเดล เช่น Dropout หรือ Batch 
Normalization จะเปลี่ยนไปตามโหมดที่โมเดลอยู่
• Dropout
ในโหมดการฝึก (train()), Dropout จะเปิดใช้งาน โดยจะสุ่มปิดบางหน่วย (neurons) ระหว่างการฝึก เพื่อช่วยป้องกันการ overfitting แต่ใน
โหมดการทดสอบ (eval()), Dropout จะถูกปิด (หรือถูกข้ามไป) และไม่มีการปิดหน่วยใดๆ
• Batch Normalization
ในโหมดการฝึก (train()), Batch Normalization จะคํานวณค่า mean และ variance จาก mini-batch ในแต่ละครั้งที่ทํา forward pass แต่ใน
โหมดการทดสอบ (eval()), Batch Normalization จะใช้ค่า mean และ variance ที่คํานวณไว้จากการฝึกเพื่อให้โมเดลทํางานคงที่ในการ
ทดสอบ
อย่างไรก็ตาม คําสั่ง model.train() 
ไม่ได้ใช้สําหรับการฝึกโมเดลโดยตรง แต่มันเป็นการบอก PyTorch 
ว่าโมเดลอยู่ในโหมดการฝึก และเมื่อ
ต้องการทดสอบโมเดลหลังจากการฝึก เราควรสลับเป็นโหมด eval() โดยใช้ model.eval() เพื่อให้แน่ใจว่าเลเยอร์ต่าง ๆ เช่น Dropout และ 
Batch Normalization ทํางานในโหมดการทดสอบอย่างถูกต้อง นอกจากนี้ หากไม่กําหนดค่า model.train() ใน PyTorch ค่าเริ่มต้น (default) 
ของโมเดลจะอยู่ใน โหมดการฝึก (training mode) โดยอัตโนมัติ ซึ่งหมายความว่า

ฟังก์ชัน optimizer.zero_grad() ใน PyTorch 
optimizer.zero_grad() ใน PyTorch เป็นคําสั่งที่ใช้ รีเซ็ต (reset) ค่า gradients ของพารามิเตอร์ทั้งหมดที่ถูกจัดการโดย optimizer ให้เป็น
ศูนย์ก่อนที่จะทําการคํานวณ gradients ใหม่ในการฝึกแต่ละรอบ (forward และ backward pass)
สาเหตุที่ต้องใช้ optimizer.zero_grad()
ใน PyTorch ค่า gradients ที่คํานวณได้จากการทําbackpropagation (ผ่าน loss.backward()) จะถูกสะสม (accumulated) ในพารามิเตอร์
ต่างๆ ของโมเดล หากไม่ได้รีเซ็ตค่า gradients เป็นศูนย์ในแต่ละรอบ ค่า gradients จากรอบก่อนหน้าจะถูกบวกเข้าไปกับค่า gradients ของรอบ
ปัจจุบัน ทําให้ค่าที่ได้ไม่ถูกต้อง ดังนั้น การใช้ optimizer.zero_grad() 
ก่อนการคํานวณ backpropagation 
ในแต่ละรอบการฝึกฝนจึงเป็น
สิ่งจําเป็นเพื่อให้แน่ใจว่าค่า gradients ใหม่จะถูกคํานวณจากข้อมูลใหม่เท่านั้น

ฟังก์ชัน criterion(...), backward(...) และ step(...) 
•
คําสั่ง loss_train = criterion(output_train, y_train) ทําหน้าที่คํานวณ ค่าความสูญเสีย (loss) ระหว่างค่าผลลัพธ์ที่โมเดลทํานาย (output_train) กับค่าจริง (y_train) โดยใช้ฟังก์ชัน
ความสูญเสียที่กําหนดโดย criterion ซึ่งเป็นขั้นตอนสําคัญในการฝึกโมเดล Machine Learning เพราะค่าความสูญเสียนี้จะถูกใช้ในการคํานวณ gradient เพื่อนําไปปรับปรุงพารามิเตอร์ของ
โมเดล
•
คําสั่งloss_train.backward() ใช้เพื่อคํานวณ gradient ของค่าความสูญเสีย (loss) ย้อนกลับไปยังพารามิเตอร์ของโมเดล ผ่านการทํางานของ backpropagation กระบวนการนี้สําคัญ
สําหรับการฝึกโมเดลเชิงลึก (deep learning) เพราะ gradients เหล่านี้จะถูกใช้ในการอัปเดตพารามิเตอร์ของโมเดล (เช่น weights และ biases) ด้วยการปรับค่าเพื่อให้โมเดลเรียนรู้จาก
ข้อมูล
•
คําสั่ง optimizer.step() ใช้สําหรับ อัปเดตพารามิเตอร์ของโมเดล โดยใช้ค่า gradients ที่ถูกคํานวณไว้ด้วย loss.backward()
กระบวนการทํางานระห่าง criterion(...), backward(...) และstep(...) 
•
ขั้นตอน Forward Pass
1.
โมเดลคํานวณค่าผลลัพธ์ (output_train) จากข้อมูลอินพุต (x_train) โดยผ่านเลเยอร์ต่างๆ ของโมเดล
2.
ค่าความสูญเสีย (loss_train) ถูกคํานวณโดยใช้ฟังก์ชันความสูญเสีย (loss function) เช่น MSELoss หรือ CrossEntropyLoss เพื่อวัดว่าผลลัพธ์ของโมเดลแตกต่างจากค่าจริง 
(target) มากแค่ไหน
•
ขั้นตอน Backward Pass
1.
เมื่อเรียก loss_train.backward() PyTorch จะคํานวณ gradients ของค่าความสูญเสียย้อนกลับไปยังพารามิเตอร์ของโมเดลทุกตัวที่กําหนดให้มี requires_grad=True
2.
ค่ากราดิเอนต์เหล่านี้จะบอกว่าแต่ละพารามิเตอร์ (เช่น weights, biases) ควรปรับอย่างไรเพื่อให้ค่าความสูญเสียลดลง
•
ขั้นตอน การอัปเดตพารามิเตอร์
1.
เมื่อทํา backward pass ผ่านการใช้ loss.backward() แล้ว ค่า gradients ของพารามิเตอร์แต่ละตัวในโมเดลจะถูกคํานวณและเก็บไว้ในตัวแปรที่เกี่ยวข้องกับพารามิเตอร์นั้นๆ
2.
คําสั่ง optimizer.step() จะใช้ค่า gradients เหล่านั้นในการปรับค่าพารามิเตอร์ (เช่น weights, biases) เพื่อให้โมเดลเรียนรู้จากข้อมูลและลดค่าความสูญเสียในรอบถัดไป

import torch.nn as nn
import torch.nn.functional as F 
# Recommend for Not Customer
class ClassificationNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        # nn.Conv2d(in_channels, out_channels, kernel_size)
        self.conv1 = nn.Conv2d(3, 6, 5) 
        self.pool = nn.MaxPool2d(2, 2)
        self.conv2 = nn.Conv2d(6, 16, 5)
        self.fc1 = nn.Linear(16 * 5 * 5, 120)
        self.fc2 = nn.Linear(120, 84)
        self.fc3 = nn.Linear(84, 10)
    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.conv2(x)))
        # flatten all dimensions except batch
        x = torch.flatten(x, 1) 
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        x = self.fc3(x)
        return x
model = ClassificationNetwork()

torch.utils. data.Dataset จากชุดคําสั่งของ Pytorch โดย
จําเป็นต้องสร้าง (Implement) 3 ฟังก์ชันเพิ่มเติม ดังนี้
import os
import pandas as pd
from torch.utils.data import Dataset
from torchvision.io import read_image
class CustomImageDataset(Dataset):
    def __init__(self, annotations_file, img_dir, transform=None, 
                 target_transform=None):
self.img_labels = pd.read_csv(annotations_file)
        self.img_dir = img_dir
        self.transform = transform
        self.target_transform = target_transform
    def __len__(self):
return len(self.img_labels)
    def __getitem__(self, idx):
        img_path = os.path.join(self.img_dir,
                                self.img_labels.iloc[idx, 0])
        image = read_image(img_path)
        label = self.img_labels.iloc[idx, 1]
        if self.transform:
            image = self.transform(image)
        if self.target_transform:
            label = self.target_transform(label)
        return image, label

คําอธิบาย
__init__
เป็นฟังก์ชันสําหรับ

(2) กําหนดวิธีการแปลงข้อมูล 
(2) กําหนดตําแหน่งของเฉลย

import torch.nn as nn
import torch.nn.functional as F
class RegressionNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        # nn.Conv2d(in_channels, out_channels, kernel_size)
        self.conv1 = nn.Conv2d(3, 6, 5) 
        self.pool = nn.MaxPool2d(2, 2)
        self.conv2 = nn.Conv2d(6, 16, 5)
        self.fc1 = nn.Linear(16 * 5 * 5, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, 1)
    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.conv2(x)))
        # flatten all dimensions except batch
        x = torch.flatten(x, 1)
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        x = self.fc3(x)
        return x
model = RegressionNetwork()

1. Creating a Custom Dataset for your files
A custom Dataset class must implement three functions: init, len, and getitem. Take a look at this implementation; the
SatelliteImageClassification images are stored in a directory img_dir, and their labels are stored separately in a CSV file
annotations_file.
__init__
The _init_ function is run once when instantiating the Dataset object. We initialize the annotations file, and both transforms
__len__
The _len_ function returns the number of samples in our dataset.
__getitem__
The _getitem_ function loads and returns a sample from the dataset at the given index idx. Based on the index, it identifies the
image’s location on disk, converts that to a tensor using read_image, retrieves the corresponding label from the csv data in
self.img_labels, calls the transform functions on them (if applicable), and returns the tensor image and corresponding label in a
tuple.
import os
import pandas as pd
from torch.utils.data import Dataset
from PIL import Image
classes = ('apple', 'avocado', 'banana', 'cherry', 'kiwi', 'mango', 'orange', 'pinenapple', 'strawberries', 'watermelon')
idx_to_class = {i:j for i, j in enumerate(classes)}
class_to_idx = {value:key for key,value in idx_to_class.items()}
class CustomImageDataset(Dataset):
   def __init__(self, annotations_file, img_dir, transform=None, target_transform=None):
       self.img_labels = pd.read_csv(annotations_file)
       self.img_dir = img_dir
       self.transform = transform
       self.target_transform = target_transform
   def __len__(self):
       return len(self.img_labels)
   def __getitem__(self, idx):
       img_path = os.path.join(self.img_dir, self.img_labels.iloc[idx, 1], self.img_labels.iloc[idx, 0])
       label = self.img_labels.iloc[idx, 1]
       # Load image
       image = Image.open(img_path).convert('RGB')
       # transform to number
       label = class_to_idx[label]
       if self.transform:
           image = self.transform(image)
       if self.target_transform:
           label = self.target_transform(label)
       return image, label
